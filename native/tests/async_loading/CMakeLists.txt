cmake_minimum_required(VERSION 3.25)
project(BetterEndfieldAsyncLoadingTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
enable_testing()
add_subdirectory(../../modules/custom_model custom_model)
add_executable(async_loading_tests scheduler_tests.cpp)
target_link_libraries(async_loading_tests PRIVATE BetterEndfield.CustomModelCore Threads::Threads)
if(MSVC)
    target_compile_options(async_loading_tests PRIVATE /W4 /utf-8 /permissive-)
else()
    target_compile_options(async_loading_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME async_loading COMMAND async_loading_tests)
add_test(NAME bem_v11_capacity COMMAND BetterEndfield.BemV11CapacityTests)
add_test(NAME bem_v13_morph COMMAND BetterEndfield.BemV13MorphTests)
