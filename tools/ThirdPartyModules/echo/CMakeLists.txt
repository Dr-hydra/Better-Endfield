cmake_minimum_required(VERSION 3.22)
project(BetterEndfieldEcho LANGUAGES CXX)
set(DEFAULT_SDK_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../../native/shared/include")
if(NOT EXISTS "${DEFAULT_SDK_INCLUDE}/BetterEndfield/ThirdPartyModule.h")
    set(DEFAULT_SDK_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
endif()
set(BETTER_ENDFIELD_SDK_INCLUDE "${DEFAULT_SDK_INCLUDE}" CACHE PATH "Directory containing BetterEndfield public ABI headers")
if(NOT EXISTS "${BETTER_ENDFIELD_SDK_INCLUDE}/BetterEndfield/ThirdPartyModule.h")
    message(FATAL_ERROR "Set BETTER_ENDFIELD_SDK_INCLUDE to the public SDK include directory")
endif()
add_library(example_echo SHARED native/echo.cpp)
target_compile_features(example_echo PRIVATE cxx_std_20)
target_include_directories(example_echo PRIVATE "${BETTER_ENDFIELD_SDK_INCLUDE}")
set_target_properties(example_echo PROPERTIES OUTPUT_NAME "example.echo")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/module.json" PACKAGE_MANIFEST)
if(WIN32)
    set_property(TARGET example_echo PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    set(PACKAGE_PLATFORM windows-x64)
    string(JSON PACKAGE_MANIFEST REMOVE "${PACKAGE_MANIFEST}" libraries android-arm64)
elseif(ANDROID AND CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
    target_link_libraries(example_echo PRIVATE c++_static c++abi)
    set(PACKAGE_PLATFORM android-arm64)
    string(JSON PACKAGE_MANIFEST REMOVE "${PACKAGE_MANIFEST}" libraries windows-x64)
    target_link_options(example_echo PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
else()
    message(FATAL_ERROR "This example supports Windows x64 and Android arm64")
endif()
set(PACKAGE_DIR "${CMAKE_BINARY_DIR}/package")
file(MAKE_DIRECTORY "${PACKAGE_DIR}")
file(WRITE "${PACKAGE_DIR}/module.json" "${PACKAGE_MANIFEST}\n")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/ui")
    file(COPY ui DESTINATION "${PACKAGE_DIR}")
endif()
set_target_properties(example_echo PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PACKAGE_DIR}/native/${PACKAGE_PLATFORM}" LIBRARY_OUTPUT_DIRECTORY "${PACKAGE_DIR}/native/${PACKAGE_PLATFORM}")
foreach(CONFIG RELEASE DEBUG RELWITHDEBINFO MINSIZEREL)
    set_target_properties(example_echo PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${CONFIG} "${PACKAGE_DIR}/native/${PACKAGE_PLATFORM}" LIBRARY_OUTPUT_DIRECTORY_${CONFIG} "${PACKAGE_DIR}/native/${PACKAGE_PLATFORM}")
endforeach()
