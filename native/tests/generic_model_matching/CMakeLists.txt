cmake_minimum_required(VERSION 3.20)
project(GenericModelMatchingChecks LANGUAGES CXX)
enable_testing()
add_executable(GenericModelMatchingTests ../generic_model_matching_tests.cpp)
add_executable(AndroidWorldBindingTests ../android_world_binding_tests.cpp)
target_include_directories(AndroidWorldBindingTests PRIVATE ../../shared/include)
foreach(target GenericModelMatchingTests AndroidWorldBindingTests)
  target_compile_features(${target} PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
  endif()
  add_test(NAME ${target} COMMAND ${target})
endforeach()
