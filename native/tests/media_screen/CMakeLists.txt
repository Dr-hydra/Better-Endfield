cmake_minimum_required(VERSION 3.20)
project(MediaScreenTests LANGUAGES C CXX)
enable_testing()
add_executable(media_screen_tests media_screen_tests.cpp)
target_compile_features(media_screen_tests PRIVATE cxx_std_20)
target_compile_definitions(media_screen_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(media_screen_tests PRIVATE ../../shared/include)
add_test(NAME media_screen COMMAND media_screen_tests)
add_executable(media_capture_ipc_tests capture_ipc_tests.cpp ../../modules/media_screen/remote_capture.cpp)
target_compile_features(media_capture_ipc_tests PRIVATE cxx_std_20)
target_compile_definitions(media_capture_ipc_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE BE_MEDIA_CAPTURE_IPC_TEST)
add_test(NAME media_screen_capture_ipc COMMAND media_capture_ipc_tests)
add_executable(media_window_input_tests window_input_tests.cpp)
target_compile_features(media_window_input_tests PRIVATE cxx_std_20)
target_compile_definitions(media_window_input_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(media_window_input_tests PRIVATE dwmapi comctl32)
add_test(NAME media_screen_window_input COMMAND media_window_input_tests)
add_subdirectory(../../shared/third_party/minhook ${CMAKE_BINARY_DIR}/game-input-minhook)
add_library(media_game_input_bridge SHARED ../../modules/media_screen/game_input_bridge.cpp)
target_compile_features(media_game_input_bridge PRIVATE cxx_std_20)
target_compile_definitions(media_game_input_bridge PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(media_game_input_bridge PRIVATE minhook dxguid)
set_target_properties(media_game_input_bridge PROPERTIES OUTPUT_NAME "BetterEndfieldNext.GameInputBridge")
if(MSVC)
    set_property(TARGET media_game_input_bridge minhook PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_executable(media_game_input_tests game_input_tests.cpp)
target_compile_features(media_game_input_tests PRIVATE cxx_std_20)
target_compile_definitions(media_game_input_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(media_game_input_tests PRIVATE dwmapi dinput8 dxguid)
add_dependencies(media_game_input_tests media_game_input_bridge)
add_test(NAME media_screen_game_input COMMAND media_game_input_tests)
