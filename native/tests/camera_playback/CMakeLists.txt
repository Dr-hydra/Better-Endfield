cmake_minimum_required(VERSION 3.25)
project(BetterEndfieldCameraPlaybackTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BE_CAMERA_SANITIZERS "Run host tests with address/undefined sanitizers" OFF)
find_package(Threads REQUIRED)
enable_testing()
foreach(name vmd path runtime character character_runtime lease actions_lease)
    add_executable(camera_${name}_tests ${name}_tests.cpp)
    target_link_libraries(camera_${name}_tests PRIVATE Threads::Threads)
    if(name STREQUAL "lease")
        target_sources(camera_${name}_tests PRIVATE ../../shared/host/pose_lease.cpp)
    endif()
    if(name STREQUAL "actions_lease")
        target_compile_definitions(camera_${name}_tests PRIVATE BE_TEST_PC_POSE)
        if(NOT WIN32)
            target_include_directories(camera_${name}_tests BEFORE PRIVATE ../../shared/android_compat/include ../../shared/android_compat)
            target_sources(camera_${name}_tests PRIVATE ../../shared/android_compat/android_win32.cpp)
        endif()
    endif()
    target_include_directories(camera_${name}_tests PRIVATE ../../shared/include)
    if(WIN32)
        target_compile_definitions(camera_${name}_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
        target_link_libraries(camera_${name}_tests PRIVATE user32)
    else()
        # Only the runtime integration harness uses this. The parser/path tests
        # have no mocked OS interfaces; they use the host's real filesystem.
        target_include_directories(camera_${name}_tests PRIVATE stubs)
    endif()
    if(MSVC)
        target_compile_options(camera_${name}_tests PRIVATE /W4 /utf-8 /permissive-)
    else()
        target_compile_options(camera_${name}_tests PRIVATE -Wall -Wextra -Wpedantic)
        if(BE_CAMERA_SANITIZERS)
            target_compile_options(camera_${name}_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(camera_${name}_tests PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
    add_test(NAME camera_${name} COMMAND camera_${name}_tests)
endforeach()
