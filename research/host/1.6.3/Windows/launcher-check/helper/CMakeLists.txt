cmake_minimum_required(VERSION 3.20)
project(EndfieldLaunchBridgeResearch LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This research helper requires Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(LaunchBridge main.cpp)
add_executable(LaunchBridgeFixture fixture.cpp)
foreach(target LaunchBridge LaunchBridgeFixture)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /EHsc)
  endif()
endforeach()
