cmake_minimum_required(VERSION 3.22)
project(BetterEndfieldHookChainTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
enable_testing()
add_executable(hook_chain_tests hook_chain_tests.cpp)
target_include_directories(hook_chain_tests PRIVATE ../../shared/include)
target_link_libraries(hook_chain_tests PRIVATE Threads::Threads)
if(WIN32)
  add_subdirectory(../../shared/third_party/minhook minhook)
  target_sources(hook_chain_tests PRIVATE ../../shared/host/hook_broker.cpp ../../shared/host/hook_diagnostics.cpp ../../shared/host/logging.cpp)
  target_link_libraries(hook_chain_tests PRIVATE minhook)
  target_compile_definitions(hook_chain_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
  target_compile_options(hook_chain_tests PRIVATE /W4 /utf-8 /permissive-)
endif()
add_test(NAME hook_chain COMMAND hook_chain_tests)
